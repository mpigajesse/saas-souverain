=== orchestrator-go/Dockerfile ===
# Image de l'orchestrateur Go (Mission C) — superviseur de failover Patroni,
# consensus etcd, serveur gRPC du contrat proto/amane/framework/v1.
#
# Contexte de build : le dossier orchestrator-go/ (module Go autonome, bindings
# gRPC déjà générés dans gen/).
#   docker build -t amane-orchestrator:dev orchestrator-go

FROM golang:1.26-alpine AS build
WORKDIR /src
COPY go.mod go.sum ./
RUN go mod download
COPY . .
# Binaire statique : l'image finale n'embarque ni libc ni toolchain.
RUN CGO_ENABLED=0 go build -trimpath -ldflags="-s -w" -o /out/orchestrator ./cmd/orchestrator

FROM alpine:3.20
# Compte non privilégié : l'orchestrateur n'a besoin d'aucun droit root
# (etcd et Patroni sont joints par le réseau).
RUN adduser -D -H -u 10001 amane
COPY --from=build /out/orchestrator /usr/local/bin/orchestrator
USER amane
EXPOSE 50051
ENTRYPOINT ["/usr/local/bin/orchestrator"]

FROM rust:1.88-bookworm AS builder
WORKDIR /workspace
# Contexte de build = spike/ (workspace Cargo complet)
COPY . .
RUN cargo build --release -p node

FROM debian:bookworm-slim
# Client PostgreSQL 17 (pg_dump / pg_restore) pour les sauvegardes chiffrées
# (refonte V3, étape 4) : il doit être de la version du serveur (Spilo 17), que
# Debian bookworm ne fournit pas — dépôt officiel PostgreSQL (PGDG), clé vérifiée.
RUN apt-get update \
 && apt-get install -y --no-install-recommends ca-certificates libssl3 curl gnupg \
 && install -d /usr/share/postgresql-common/pgdg \
 && curl -fsSL -o /usr/share/postgresql-common/pgdg/apt.postgresql.org.asc https://www.postgresql.org/media/keys/ACCC4CF8.asc \
 && echo "deb [signed-by=/usr/share/postgresql-common/pgdg/apt.postgresql.org.asc] https://apt.postgresql.org/pub/repos/apt bookworm-pgdg main" > /etc/apt/sources.list.d/pgdg.list \
 && apt-get update \
 && apt-get install -y --no-install-recommends postgresql-client-17 \
 && apt-get purge -y curl gnupg && apt-get autoremove -y \
 && rm -rf /var/lib/apt/lists/*
COPY --from=builder /workspace/target/release/ss-node /usr/local/bin/ss-node
ENV SS_NODE_DIR=/data
VOLUME ["/data"]
EXPOSE 9001

# Script d'entrée : init --first si premier serveur (génère la DEK), init seul
# sinon (le serveur attend son enrôlement : `ss-node enroler`).
RUN printf '#!/bin/sh\nset -e\nif [ ! -f "$SS_NODE_DIR/config.toml" ]; then\n  if [ "${NODE_MODE}" = "active" ]; then\n    ss-node init --first\n  else\n    ss-node init\n  fi\nfi\nexec ss-node "$@"\n' > /entrypoint.sh && chmod +x /entrypoint.sh

ENTRYPOINT ["/entrypoint.sh"]
CMD ["run", "--mode", "passive"]

54-## 3. Garanties
55-
--
58-- **Sauvegardes** : `pg_dump` chiffré par la DEK, déposé sur le relais en blob opaque, rotation sur 7 jours, restauration par `ss-node restaurer`.
59-- **Chiffrement au repos** : délégué au chiffrement du disque (BitLocker, LUKS), vérifié par l'installeur et signalé à la PME.
60-- **Haute disponibilité** : etcd + Patroni, aucun consensus écrit à la main. Bascule automatique avec un quorum de trois membres etcd ; synchrone strict à partir de trois serveurs PostgreSQL ; fencing par la timeline PostgreSQL ; alerte au portail sous le seuil (ADR-004, ADR-006).
61:- **Images** : la PME exécute exactement ce qui a été validé (digest), et une application n'est validée que si son image est publiée dans le registre (ADR-009).
